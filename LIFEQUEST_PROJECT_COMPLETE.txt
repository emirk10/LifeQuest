================================================================================
 1. PROJE ÖZETİ
================================================================================

LifeQuest, kullanıcıların kişisel gelişim hedeflerini başarmalarına yardımcı
olan, yapay zeka ve oyunlaştırma mekanizmalarından yararlanan modern bir
platformdur.

📌 Temel Amaçlar:
  • Kullanıcıları günlük görevler tamamlamaya motive etmek
  • Kategori bazlı (Eğitim, Sağlık, Kariyer, Sosyal) seviye sistemi
  • XP (Deneyim Puanı) kazanarak seviye atlama
  • Rozetler ve başarılar açma
  • Sosyal özellikler (arkadaşlık, yarışma)
  • AI asistanı ile kişiselleştirilmiş görev önerileri

💡 Kişiye Özgü Yaklaşım:
  • Kayıt sırasında "Seni Tanıyalım" anketi
  • Kullanıcı profiline göre akıllı görev atama
  • Timer ve basit doğrulama ile gerçekçi görev tamamlama

3.3 REACTİON FLOW

Kayıt Akışı:
  Register.jsx → Anket Soruları Göster
             → Anket Cevapları Topla
             → POST /api/auth/register (+ questionnaire data)
             → Backend: AppUser.Goals, DailyAvailableMinutes vb. kaydet
             → Token al → Login sayfasına yönlendir

Dashboard Akışı:
  DashBoard.jsx → Yükleniyor...
             → GET /api/users/{userId} (İstatistikler)
             → GET /api/user-tasks/get-recommendations (YENİ - 3 görev öner)
             → Önerilen görevleri göster
             → Kullanıcı "Seç" tıklayınca
             → POST /api/user-tasks/request-task
             → TaskList.jsx'e yönlendir

  - loading: boolean

Features:
  - 3 görev kartı göster
  - Her kartında: Başlık, Açıklama, XP, Zorluk, Match Score (%)
  - "Seç" butonu

──────────────────────

  - taskTitle: string
  - onComplete: function

Features:
  - Countdown timer

// XP ve Level Hesaplaması
public int CalculateXP(UserTask completedTask)
{
    int baseXP = completedTask.AppTask.BaseXP;
    double difficultyMultiplier = completedTask.AppTask.Difficulty == DifficultyLevel.Hard ? 1.5 :
                                  completedTask.AppTask.Difficulty == DifficultyLevel.Medium ? 1.2 : 1.0;
    return (int)(baseXP * difficultyMultiplier);
}

public void GrantXP(AppUser user, int xp)
{
    user.TotalXP += xp;
    
    // Seviye Sistemi: Her 1000 XP'de 1 seviye
    double newLevel = Math.Floor(user.TotalXP / 1000.0) + 1;
    if (newLevel > user.GeneralLevel)
    {
        user.GeneralLevel = newLevel;
        // LevelUp bildirim gönder
        _notificationService.SendLevelUpNotification(user);
    }
}

// ML.NET model kullanarak görev önerme
public async Task<List<Guid>> GetRecommendedTasksAsync(Guid userId, int count = 3)
{
    var user = await _userRepository.GetByIdAsync(userId);
    var userProfile = MapToUserProfile(user); // Questionnaire verileri
    var availableTasks = await _taskRepository.GetAvailableTasksAsync();
    
    // ML Model'i kullan veya Kural Tabanlı Algoritma
    var scoredTasks = availableTasks.Select(task => new {
        TaskId = task.Id,
        Score = CalculateMatchScore(userProfile, task)
    })
    .OrderByDescending(x => x.Score)
    .Take(count)
    .Select(x => x.TaskId)
    .ToList();
    
    return scoredTasks;
}

private float CalculateMatchScore(UserProfile profile, AppTask task)
{
    float score = 0;
    
    // 1. Goal Match (0.4 weight)
    if (profile.Goals.Contains(task.CategoryName.ToLower()))
        score += 0.4f;
    
    // 2. Strength Match (0.25 weight)
    if (profile.Strengths.Any(s => task.Description.Contains(s)))
        score += 0.25f;
    
    // 3. Time Match (0.2 weight)
    if (task.EstimatedMinutes <= profile.DailyAvailableMinutes)
        score += 0.2f;
    
    // 4. Difficulty Match (0.15 weight)
    if (IsMatchingDifficulty(task, profile.DifficultyPreference))
        score += 0.15f;
    
    return score;
}

// Her gün 00:00'de çalışır, her kullanıcıya 3 görev atar
public class DailyQuestWorker : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            var nextRun = DateTime.Today.AddDays(1); // Yarın 00:00
            var delay = nextRun - DateTime.Now;
            
            await Task.Delay(delay, stoppingToken);
            
            // Tüm aktif kullanıcılara görev ata
            var users = await _userRepository.GetAllActiveUsersAsync();
            foreach (var user in users)
            {
                await _dailyQuestService.AssignDailyQuestsAsync(user.Id, 3);
            }
        }
    }
}

public class TaskRecommendationService : ITaskRecommendationService
{
    public async Task<List<Guid>> GetRecommendedTasksAsync(
        Guid userId, 
        int count = 3)
    {
        var user = await _userRepository.GetByIdAsync(userId);
        var availableTasks = await _taskRepository.GetActiveTasksAsync();
        
        var scored = availableTasks.Select(task => new {
            TaskId = task.Id,
            Score = CalculateScore(user, task)
        })
        .OrderByDescending(x => x.Score)
        .Take(count)
        .Select(x => x.TaskId)
        .ToList();
        
        return scored;
    }
    
    private float CalculateScore(AppUser user, AppTask task)
    {
        float score = 0;
        
        // Goal Match (40%)
        if (user.Goals?.Contains(task.Category.Name.ToLower()) == true)
            score += 0.4f;
        
        // Strength Match (25%)
        if (user.Strengths?.Any(s => task.Description.Contains(s)) == true)
            score += 0.25f;
        
        // Time Match (20%)
        if (task.EstimatedMinutes <= user.DailyAvailableMinutes)
            score += 0.2f;
        
        // Difficulty Match (15%)
        bool difficultyMatch = user.DifficultyPreference == "mixed" ||
                               (user.DifficultyPreference == "easy" && task.Difficulty == DifficultyLevel.Easy) ||
                               (user.DifficultyPreference == "medium" && task.Difficulty == DifficultyLevel.Medium) ||
                               (user.DifficultyPreference == "hard" && task.Difficulty == DifficultyLevel.Hard);
        if (difficultyMatch)
            score += 0.15f;
        
        return score;
    }
}

 12. AUTHENTICATION & SECURITY
================================================================================

JWT Token:
──────────
- Algorithm: HS256 (SHA-256)
- Secret Key: 64+ karakterli key (appsettings.json)
- Expiration: Configurable (default: 24 saat)
- Claims: userId, role

Token Storage (Frontend):
─────────────────────────
- localStorage'da sakla
- Her request'te Authorization header'a ekle
- 401 response'da yeni token al (refresh)

Rate Limiting:
──────────────
- 500 requests / 1 minute
- Per user (userId) + IP basis
- 429 (Too Many Requests) response


CORS:
─────
- AllowedOrigins: localhost (dev), production domain (prod)
- AllowedMethods: GET, POST, PUT, PATCH, DELETE, OPTIONS
- AllowedHeaders: *, Authorization, Content-Type
- AllowCredentials: true (SignalR için)
